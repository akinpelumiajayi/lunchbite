# Runtime dependencies for the LunchBite dashboard (`streamlit run app/LunchBite.py`).
#
# This file is what Streamlit Community Cloud installs, so it carries only what
# the app itself needs. The notebook, plotting and test dependencies moved to
# requirements-dev.txt — Jupyter alone pulls ~40 packages the deployed app never
# imports, and the cloud build has a disk and time budget.
#
#   local development:   pip install -r requirements.txt -r requirements-dev.txt
#   reproduce a result:  pip install -r requirements.lock.txt
#
# requirements.lock.txt still pins the exact versions the published benchmark
# results were produced with, which are well ahead of the minimums below.

# ── PyTorch: CPU build ───────────────────────────────────────────────────────
# sentence-transformers pulls torch in either way. Declared here with the CPU
# index in front so the resolver takes the ~200 MB CPU wheel rather than PyPI's
# default Linux wheel, which bundles the CUDA runtime and is several times the
# size — enough on its own to fail a Streamlit Cloud build on disk. The `+cpu`
# local version sorts above the plain one, so the CPU index wins the resolve.
# Streamlit Cloud has no GPU, so nothing is given up.
--extra-index-url https://download.pytorch.org/whl/cpu
torch>=2.2.0

# ── Environment ───────────────────────────────────────────────────────────────
python-dotenv>=1.0.0

# ── Core RAG & vector store ───────────────────────────────────────────────────
chromadb>=0.5.0
rank-bm25>=0.2.2

# ── LangChain / LangGraph / LangSmith ────────────────────────────────────────
langchain>=1.3.0
langchain-core>=0.3.0
langchain-groq>=0.3.0
# Kept for the deployed app even though Ollama is unreachable from the cloud:
# llm_provider imports it inside _try_ollama, which the resolver still reaches
# when GROQ_API_KEY is absent. Without it that path raises ImportError instead
# of the "not reachable" message the app knows how to display.
langchain-ollama>=0.2.0
langgraph>=0.3.0
langsmith>=0.1.0

# ── LLM providers (Groq and/or Ollama — NO Anthropic) ────────────────────────
# Groq (cloud, free tier): get key at https://console.groq.com
groq>=0.4.0
# Ollama (local, no key): install from https://ollama.com, then: ollama pull llama3.2

# ── Retrieval models (REQUIRED) ──────────────────────────────────────────────
# Semantic retrieval uses all-MiniLM-L6-v2 and reranking uses
# ms-marco-MiniLM-L-6-v2. Both download on first run (~90 MB each) and are
# cached in ~/.cache/huggingface/. This is no longer an optional upgrade path:
# the previous offline TF-IDF embedding backend has been removed.
sentence-transformers>=3.0.0

# ── Statistics ───────────────────────────────────────────────────────────────
# Imported directly by src/reward/agreement.py. They already arrive as
# transitive dependencies of sentence-transformers, which is exactly why they are
# declared here: a future sentence-transformers release that drops either one
# would break `annotate.py --report` with no signal from this file.
#   scipy      1.9 is the floor: spearmanr() gained the .statistic attribute there
#   sklearn    cohen_kappa_score(weights="quadratic")
scipy>=1.9.0
scikit-learn>=1.3.0

# ── Dashboard (app/LunchBite.py) ─────────────────────────────────────────────
# pandas and altair arrive with it, so the dashboard's charts and tables add no
# further declared dependency. Not needed by the benchmark, the eval harness or
# the CLI — nothing under src/ imports it.
#   1.40 is the floor: st.toggle, and `width=` on button/dataframe/altair_chart.
streamlit>=1.40.0

# ── Utilities ─────────────────────────────────────────────────────────────────
numpy>=1.26.0
# TypedDict with the total=False semantics src/graphs/state.py relies on.
typing-extensions>=4.0.0
